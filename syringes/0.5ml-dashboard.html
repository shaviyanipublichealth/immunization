function doGet() { return HtmlService.createHtmlOutputFromFile('Page') .setTitle('HCF Inventory Dashboard') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.DEFAULTALLOWALL); } function getSheetData() { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; var headers = rows[0]; var data = []; for (var i = 1; i < rows.length; i++) { var row = rows[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { var headerName = headers[j] ? headers[j].toString().trim() : ''; if (!headerName) continue; var val = row[j]; if (val instanceof Date) { val = Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); } obj[headerName] = val; } data.push(obj); } return data; } catch (err) { throw new Error('Failed to fetch sheet data: ' + err.message); } }
